#include "stability.hpp"
#include <cmath>
#include <algorithm>
#include <complex>

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

bool checkDiscreteStability(const std::vector<double>& coeffs) {
    int n = static_cast<int>(coeffs.size()) - 1;
    if (n <= 0) return true;

    // Íîðìèðîâêà ïîëèíîìà, ÷òîáû ñòàðøèé êîýôôèöèåíò áûë ðàâåí 1.0
    std::vector<double> p(coeffs.size());
    double lead = coeffs.back();
    if (std::abs(lead) < 1e-14) return false;
    for (size_t i = 0; i < coeffs.size(); ++i) {
        p[i] = coeffs[i] / lead;
    }

    // Èíèöèàëèçàöèÿ ïðèáëèæåííûõ êîðíåé (ìåòîä Äþðàí-Êåðíåðà)
    std::vector<std::complex<double>> roots(n);
    double r = 0.4;
    for (int i = 0; i < n; ++i) {
        double angle = 2.0 * M_PI * i / n + 0.3;
        roots[i] = std::complex<double>(r * std::cos(angle), r * std::sin(angle));
    }

    // Èòåðàöèîííûé ïðîöåññ óòî÷íåíèÿ êîðíåé ïîëèíîìà
    const int max_iter = 200;
    for (int iter = 0; iter < max_iter; ++iter) {
        double max_delta = 0.0;
        for (int i = 0; i < n; ++i) {
            std::complex<double> num = p[n];
            for (int j = n - 1; j >= 0; --j) {
                num = num * roots[i] + p[j];
            }

            std::complex<double> den = 1.0;
            for (int j = 0; j < n; ++j) {
                if (i != j) {
                    den *= (roots[i] - roots[j]);
                }
            }

            if (std::abs(den) < 1e-14) continue;
            std::complex<double> delta = num / den;
            roots[i] -= delta;
            max_delta = std::max(max_delta, std::abs(delta));
        }
        if (max_delta < 1e-10) break;
    }

    // Ïðîâåðêà óñëîâèÿ: ìîäóëü êàæäîãî êîðíÿ äîëæåí áûòü ñòðîãî ìåíüøå 1.0
    const double EPS = 1e-6;
    for (int i = 0; i < n; ++i) {
        if (std::abs(roots[i]) >= (1.0 - EPS)) {
            return false; // Íàøåëñÿ êîðåíü âíå åäèíè÷íîé îêðóæíîñòè -> íåóñòîé÷èâî
        }
    }

    return true;
}
