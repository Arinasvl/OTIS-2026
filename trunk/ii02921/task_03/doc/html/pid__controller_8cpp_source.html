39double PIDController::calculate(double error, double dt)
40{
41 if (dt <= 0.0)
42 {
43 throw std::invalid_argument(
44 "Time step must be greater than zero"
45 );
46 }
47
48 // Пропорциональная составляющая.
49 double deltaP =
50 m_kp * (error - m_previousError);
51
52 // Интегральная составляющая.
53 double deltaI =
54 m_ki * error * dt;
55
56 // Дифференциальная составляющая.
57 double deltaD =
58 (m_kd / dt) *
59 (error - 2.0 * m_previousError + m_previousError2);
60
61 // Предварительное новое управляющее воздействие.
62 double newOutput =
63 m_previousOutput + deltaP + deltaI + deltaD;
64
65 // Антиwindup: не допускаем накопления интегрального
66 // воздействия, если оно усиливает насыщение регулятора.
67 if (newOutput > m_maxOutput && deltaI > 0.0)
68 {
69 deltaI = 0.0;
70 }
71 else if (newOutput < m_minOutput && deltaI < 0.0)
72 {
73 deltaI = 0.0;
74 }
75
76 // Пересчитываем управляющее воздействие.
77 newOutput =
78 m_previousOutput + deltaP + deltaI + deltaD;
79
80 // Ограничиваем выходной сигнал.
81 newOutput = std::max(m_minOutput, newOutput);
82 newOutput = std::min(m_maxOutput, newOutput);
83
84 // Обновляем историю ошибок.
85 m_previousError2 = m_previousError;
86 m_previousError = error;
87
88 // Сохраняем текущее управляющее воздействие.
89 m_previousOutput = newOutput;
90
91 return newOutput;
92}
